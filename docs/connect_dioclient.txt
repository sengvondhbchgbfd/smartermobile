 Override — when you need to swap per feature

 // Step 1: stub in feature
final featureDioClientProvider = Provider<DioClient>(
  (ref) => throw UnimplementedError(),
);

// Step 2: register in main.dart
final container = ProviderContainer(
  overrides: [
    featureDioClientProvider.overrideWith(
      (ref) => ref.watch(dioClientProvider).requireValue,
    ),
  ],
);

 final dioClientFamilyProvider = Provider.family<DioClient, String>(

// usage
final client = ref.watch(dioClientFamilyProvider('https://payment.api.com'));
